Python Basics

Fundamental Python syntax and concepts.

# Variables and data types
name = "Alice"
age = 30
height = 5.7
is_student = True

# Output
print("Hello, World!")
print(f"Name: {name}, Age: {age}")

# Input
user_input = input("Enter your name: ")
print(f"Hello, {user_input}!")

# Arithmetic operations
a = 10
b = 3
print(f"a + b = {a + b}")
print(f"a - b = {a - b}")
print(f"a * b = {a * b}")
print(f"a / b = {a / b}")
print(f"a // b = {a // b}")  # Floor division
print(f"a % b = {a % b}")    # Modulus
print(f"a ** b = {a ** b}")  # Exponentiation

# String operations
greeting = "Hello"
name = "World"
message = greeting + " " + name
print(message)
print(message.upper())
print(message.lower())
print(message.replace("World", "Python"))

# Conditional statements
score = 85
if score >= 90:
    grade = "A"
elif score >= 80:
    grade = "B"
elif score >= 70:
    grade = "C"
else:
    grade = "F"
print(f"Score: {score}, Grade: {grade}")

# Loops
# For loop
print("For loop:")
for i in range(5):
    print(i)

# While loop
print("While loop:")
count = 0
while count < 5:
    print(count)
    count += 1

# Functions
def greet_person(name, greeting="Hello"):
    return f"{greeting}, {name}!"

message = greet_person("Alice")
print(message)
message = greet_person("Bob", "Hi")
print(message)

# Lambda functions
square = lambda x: x * x
print(f"Square of 5: {square(5)}")

# List comprehensions
squares = [x**2 for x in range(10)]
print(f"Squares: {squares}")

Lists and Tuples

Working with Python's primary sequence types.

# Lists (mutable)
fruits = ["apple", "banana", "cherry", "date"]
numbers = [1, 2, 3, 4, 5]
mixed = [1, "hello", 3.14, True]

# Accessing elements
print(fruits[0])       # First element
print(fruits[-1])      # Last element
print(fruits[1:3])     # Slice from index 1 to 2
print(fruits[:2])      # Slice from start to index 1
print(fruits[2:])      # Slice from index 2 to end

# Modifying lists
fruits.append("elderberry")        # Add to end
fruits.insert(1, "blueberry")      # Insert at index
fruits[0] = "avocado"              # Modify element
fruits.remove("banana")            # Remove by value
popped = fruits.pop()              # Remove and return last element
popped_idx = fruits.pop(1)         # Remove and return element at index

# List operations
more_fruits = ["fig", "grape"]
all_fruits = fruits + more_fruits  # Concatenation
repeated = numbers * 2             # Repetition

# List methods
print(len(fruits))                 # Length
print("apple" in fruits)           # Membership check
fruits.sort()                      # Sort in place
fruits.reverse()                   # Reverse in place
sorted_fruits = sorted(fruits)     # Return new sorted list

# List comprehensions
squares = [x**2 for x in range(10)]
even_squares = [x**2 for x in range(10) if x % 2 == 0]

# Tuples (immutable)
coordinates = (10, 20)
rgb = (255, 128, 0)
person = ("Alice", 30, "Engineer")

# Accessing tuple elements
print(coordinates[0])
print(coordinates[1])

# Tuples can't be modified
# coordinates[0] = 15  # This would raise an error

# Tuple unpacking
x, y = coordinates
red, green, blue = rgb
name, age, profession = person

# Functions returning multiple values
def get_stats(numbers):
    return min(numbers), max(numbers), sum(numbers)/len(numbers)

min_val, max_val, avg_val = get_stats([1, 2, 3, 4, 5])

# Using tuples as keys in dictionaries
location_map = {
    (40.7128, -74.0060): "New York",
    (34.0522, -118.2437): "Los Angeles"
}

# Zip function
names = ["Alice", "Bob", "Charlie"]
ages = [25, 30, 35]
people = list(zip(names, ages))
print(people)

Dictionaries

Working with key-value pairs in Python.

# Creating dictionaries
person = {
    "name": "Alice",
    "age": 30,
    "city": "New York",
    "is_student": False
}

# Alternative ways to create dictionaries
person2 = dict(name="Bob", age=25, city="Boston")
person3 = dict([("name", "Charlie"), ("age", 35), ("city", "Chicago")])

# Accessing values
print(person["name"])          # Key access
print(person.get("age"))       # Using get method
print(person.get("country", "USA"))  # Default value if key doesn't exist

# Modifying dictionaries
person["age"] = 31             # Update value
person["country"] = "USA"      # Add new key-value pair
person.update({"city": "Brooklyn", "is_employed": True})  # Multiple updates

# Removing items
email = person.pop("email", None)  # Remove key and return value, default if not found
del person["is_student"]       # Remove key
person.clear()                 # Remove all items

# Dictionary methods
print(person.keys())           # View of all keys
print(person.values())         # View of all values
print(person.items())          # View of all key-value pairs

# Iterating through dictionaries
for key in person:
    print(f"{key}: {person[key]}")

for key, value in person.items():
    print(f"{key}: {value}")

# Dictionary comprehensions
squares = {x: x**2 for x in range(1, 6)}
even_squares = {x: x**2 for x in range(1, 11) if x % 2 == 0}

# Nested dictionaries
users = {
    "alice": {
        "age": 30,
        "email": "alice@example.com",
        "roles": ["admin", "user"]
    },
    "bob": {
        "age": 25,
        "email": "bob@example.com",
        "roles": ["user"]
    }
}

# Accessing nested data
print(users["alice"]["email"])
print(users["bob"]["roles"][0])

# Default dictionaries
from collections import defaultdict

word_count = defaultdict(int)
for word in ["apple", "banana", "apple", "cherry", "banana", "apple"]:
    word_count[word] += 1

# Counter
from collections import Counter

words = ["apple", "banana", "apple", "cherry", "banana", "apple"]
word_count = Counter(words)
print(word_count.most_common(2))

# Merging dictionaries (Python 3.9+)
dict1 = {"a": 1, "b": 2}
dict2 = {"b": 3, "c": 4}
merged = dict1 | dict2          # {'a': 1, 'b': 3, 'c': 4}

# Dictionary views
keys_view = person.keys()
values_view = person.values()
items_view = person.items()

Object-Oriented Programming

Creating classes and objects in Python.

# Basic class
class Dog:
    # Class attribute
    species = "Canis familiaris"
    
    # Initializer / Instance attributes
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    # Instance method
    def description(self):
        return f"{self.name} is {self.age} years old"
    
    # Another instance method
    def speak(self, sound):
        return f"{self.name} says {sound}"

# Creating objects
buddy = Dog("Buddy", 9)
miles = Dog("Miles", 4)

# Accessing attributes and methods
print(buddy.name)
print(buddy.age)
print(buddy.description())
print(buddy.speak("Woof Woof"))

# Inheritance
class Bulldog(Dog):
    def run(self, speed):
        return f"{self.name} runs at {speed} mph"

class Dachshund(Dog):
    pass

class RussellTerrier(Dog):
    pass

# Child classes inherit attributes and methods
jim = Bulldog("Jim", 5)
print(jim.description())
print(jim.run(12))

# Method overriding
class Bulldog(Dog):
    def speak(self, sound="Arf"):
        return f"{self.name} says {sound}"

jim = Bulldog("Jim", 5)
print(jim.speak())

# isinstance() and issubclass()
print(isinstance(jim, Dog))        # True
print(isinstance(jim, Bulldog))    # True
print(issubclass(Bulldog, Dog))    # True

# Properties
class Person:
    def __init__(self, first_name, last_name):
        self.first_name = first_name
        self.last_name = last_name
    
    @property
    def full_name(self):
        return f"{self.first_name} {self.last_name}"
    
    @full_name.setter
    def full_name(self, name):
        first, last = name.split(" ")
        self.first_name = first
        self.last_name = last

person = Person("John", "Doe")
print(person.full_name)            # John Doe
person.full_name = "Jane Smith"
print(person.first_name)           # Jane
print(person.last_name)            # Smith

# Class methods and static methods
class MyClass:
    class_attribute = "I'm a class attribute"
    
    def __init__(self, value):
        self.instance_attribute = value
    
    @classmethod
    def class_method(cls):
        return f"Called class method of {cls}"
    
    @staticmethod
    def static_method():
        return "Called static method"
    
    def instance_method(self):
        return f"Called instance method of {self}"

obj = MyClass("test")
print(MyClass.class_method())
print(MyClass.static_method())
print(obj.instance_method())

# Special methods
class Book:
    def __init__(self, title, author, pages):
        self.title = title
        self.author = author
        self.pages = pages
    
    def __str__(self):
        return f"{self.title} by {self.author}"
    
    def __len__(self):
        return self.pages
    
    def __eq__(self, other):
        return self.title == other.title and self.author == other.author

book1 = Book("Python Basics", "John Doe", 200)
book2 = Book("Python Basics", "John Doe", 250)

print(str(book1))      # Python Basics by John Doe
print(len(book1))      # 200
print(book1 == book2)  # True

# Abstract base classes
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self):
        pass
    
    @abstractmethod
    def perimeter(self):
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)

# This would raise an error:
# shape = Shape()

rect = Rectangle(5, 3)
print(rect.area())         # 15
print(rect.perimeter())    # 16
