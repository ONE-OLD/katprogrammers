    cout << "a * b = " << (a * b) << endl;
    cout << "a / b = " << (a / b) << endl;
    cout << "a % b = " << (a % b) << endl;

    // Conditional statements
    if (a > b) {
        cout << "a is greater than b" << endl;
    } else if (a < b) {
        cout << "a is less than b" << endl;
    } else {
        cout << "a is equal to b" << endl;
    }

    // Switch statement
    int day = 3;
    switch (day) {
        case 1: cout << "Monday" << endl; break;
        case 2: cout << "Tuesday" << endl; break;
        case 3: cout << "Wednesday" << endl; break;
        default: cout << "Other day" << endl;
    }

    // Loops
    // For loop
    cout << "For loop: ";
    for (int i = 0; i < 5; i++) {
        cout << i << " ";
    }
    cout << endl;

    // While loop
    cout << "While loop: ";
    int count = 0;
    while (count < 5) {
        cout << count << " ";
        count++;
    }
    cout << endl;

    // Do-while loop
    cout << "Do-while loop: ";
    count = 0;
    do {
        cout << count << " ";
        count++;
    } while (count < 5);
    cout << endl;

    return 0;
}

Functions in C++

Creating and using functions in C++.

#include 
#include 
using namespace std;

// Function declaration (prototype)
double calculateCircleArea(double radius);

// Function with default parameter
void greet(string name = "Guest") {
    cout << "Hello, " << name << "!" << endl;
}

// Function overloading
int add(int a, int b) {
    return a + b;
}

double add(double a, double b) {
    return a + b;
}

// Pass by value
void incrementByValue(int x) {
    x++;
    cout << "Inside function (by value): " << x << endl;
}

// Pass by reference
void incrementByReference(int &x) {
    x++;
    cout << "Inside function (by reference): " << x << endl;
}

// Pass by pointer
void incrementByPointer(int *x) {
    (*x)++;
    cout << "Inside function (by pointer): " << *x << endl;
}

// Return by value
int square(int x) {
    return x * x;
}

// Return by reference (caution: don't return local variables)
int& getElement(int arr[], int index) {
    return arr[index];
}

// Inline function
inline int cube(int x) {
    return x * x * x;
}

// Recursive function
int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

// Function template
template 
T getMax(T a, T b) {
    return (a > b) ? a : b;
}

// Main function
int main() {
    // Basic function call
    double area = calculateCircleArea(5.0);
    cout << "Area of circle: " << area << endl;

    // Default parameter
    greet(); // Uses default
    greet("Alice"); // Uses provided value

    // Function overloading
    cout << "Int addition: " << add(5, 3) << endl;
    cout << "Double addition: " << add(5.5, 3.2) << endl;

    // Parameter passing
    int num = 10;
    cout << "Original value: " << num << endl;
    
    incrementByValue(num);
    cout << "After pass by value: " << num << endl;
    
    incrementByReference(num);
    cout << "After pass by reference: " << num << endl;
    
    incrementByPointer(&num);
    cout << "After pass by pointer: " << num << endl;

    // Return values
    cout << "Square of 7: " << square(7) << endl;

    int numbers[] = {10, 20, 30};
    getElement(numbers, 1) = 25; // Modify array element
    cout << "Modified array element: " << numbers[1] << endl;

    // Inline function
    cout << "Cube of 4: " << cube(4) << endl;

    // Recursive function
    cout << "Factorial of 5: " << factorial(5) << endl;

    // Function template
    cout << "Max of 5 and 8: " << getMax(5, 8) << endl;
    cout << "Max of 3.14 and 2.71: " << getMax(3.14, 2.71) << endl;
    cout << "Max of 'A' and 'Z': " << getMax('A', 'Z') << endl;

    return 0;
}

    return M_PI * radius * radius;

Classes and Objects

Object-oriented programming with classes in C++.

#include 
#include 
using namespace std;

// Simple class
class Rectangle {
private:
    double length;
    double width;
    
public:
    // Constructor
    Rectangle(double l, double w) : length(l), width(w) {}
    
    // Default constructor
    Rectangle() : length(0), width(0) {}
    
    // Destructor
    ~Rectangle() {
        cout << "Rectangle object destroyed" << endl;
    }
    
    // Getter methods
    double getLength() const { return length; }
    double getWidth() const { return width; }
    
    // Setter methods
    void setLength(double l) { length = l; }
    void setWidth(double w) { width = w; }
    
    // Member function
    double area() const {
        return length * width;
    }
    
    double perimeter() const {
        return 2 * (length + width);
    }
};

// Class with inheritance
class Shape {
protected:
    string name;
    
public:
    Shape(string n) : name(n) {}
    
    virtual double area() const = 0; // Pure virtual function
    virtual double perimeter() const = 0;
    
    void display() const {
        cout << "Shape: " << name << endl;
    }
    
    virtual ~Shape() {} // Virtual destructor
};

class Circle : public Shape {
private:
    double radius;
    
public:
    Circle(double r) : Shape("Circle"), radius(r) {}
    
    double area() const override {
        return 3.14159 * radius * radius;
    }
    
    double perimeter() const override {
        return 2 * 3.14159 * radius;
    }
    
    double getRadius() const { return radius; }
    void setRadius(double r) { radius = r; }
};

// Class with static members
class Counter {
private:
    static int count; // Static member variable
    
public:
    Counter() {
        count++; // Increment count when object is created
    }
    
    ~Counter() {
        count--; // Decrement count when object is destroyed
    }
    
    static int getCount() { // Static member function
        return count;
    }
};

// Initialize static member
int Counter::count = 0;

// Class template
template 
class Box {
private:
    T content;
    
public:
    Box(T c) : content(c) {}
    
    T getContent() const { return content; }
    void setContent(T c) { content = c; }
};

// Main function
int main() {
    // Create objects
    Rectangle rect(5.0, 3.0);
    cout << "Rectangle area: " << rect.area() << endl;
    cout << "Rectangle perimeter: " << rect.perimeter() << endl;
    
    // Using getters and setters
    rect.setLength(7.0);
    rect.setWidth(4.0);
    cout << "Updated area: " << rect.area() << endl;
    
    // Inheritance example
    Circle circle(5.0);
    circle.display();
    cout << "Circle area: " << circle.area() << endl;
    cout << "Circle circumference: " << circle.perimeter() << endl;
    
    // Polymorphism
    Shape* shape = &circle;
    cout << "Area via base pointer: " << shape->area() << endl;
    
    // Static members
    cout << "Initial count: " << Counter::getCount() << endl;
    
    Counter c1;
    cout << "After creating c1: " << Counter::getCount() << endl;
    
    {
        Counter c2;
        cout << "After creating c2: " << Counter::getCount() << endl;
    }
    
    cout << "After c2 is destroyed: " << Counter::getCount() << endl;
    
    // Class template
    Box intBox(42);
    Box stringBox("Hello, Template!");
    
    cout << "Int box content: " << intBox.getContent() << endl;
    cout << "String box content: " << stringBox.getContent() << endl;
    
    return 0;
}

Standard Template Library (STL)

Using the powerful STL containers and algorithms.

#include 
#include 
#include 
#include 
#include 
#include 
#include 
using namespace std;

// Function to print vector elements
template 
void printVector(const vector& vec) {
    for (const auto& element : vec) {
        cout << element << " ";
    }
    cout << endl;
}

// Function to print map elements
template 
void printMap(const map& m) {
    for (const auto& pair : m) {
        cout << pair.first << ": " << pair.second << endl;
    }
}

int main() {
    // VECTOR
    cout << "=== VECTOR ===" << endl;
    vector numbers = {5, 2, 8, 1, 9};
    
    // Add elements
    numbers.push_back(7);
    numbers.insert(numbers.begin(), 0);
    
    cout << "Vector elements: ";
    printVector(numbers);
    
    // Access elements
    cout << "First element: " << numbers.front() << endl;
    cout << "Last element: " << numbers.back() << endl;
    cout << "Element at index 2: " << numbers[2] << endl;
    
    // Size and capacity
    cout << "Size: " << numbers.size() << endl;
    cout << "Capacity: " << numbers.capacity() << endl;
    
    // Sort vector
    sort(numbers.begin(), numbers.end());
    cout << "Sorted vector: ";
    printVector(numbers);
    
    // Reverse vector
    reverse(numbers.begin(), numbers.end());
    cout << "Reversed vector: ";
    printVector(numbers);
    
    // LIST
    cout << "\n=== LIST ===" << endl;
    list names = {"Alice", "Bob", "Charlie"};
    
    // Add elements
    names.push_back("David");
    names.push_front("Zoe");
    
    // Iterate through list
    cout << "List elements: ";
    for (const auto& name : names) {
        cout << name << " ";
    }
    cout << endl;
    
    // MAP
    cout << "\n=== MAP ===" << endl;
    map ageMap;
    
    // Add elements
    ageMap["Alice"] = 25;
    ageMap["Bob"] = 30;
    ageMap["Charlie"] = 35;
    
    // Access elements
    cout << "Alice's age: " << ageMap["Alice"] << endl;
    
    // Check if key exists
    if (ageMap.find("David") == ageMap.end()) {
        cout << "David not found in map" << endl;
    }
    
    // Print map
    cout << "Age map:" << endl;
    printMap(ageMap);
    
    // SET
    cout << "\n=== SET ===" << endl;
    set uniqueNumbers = {5, 2, 8, 2, 5, 9, 1};
    
    // Add elements
    uniqueNumbers.insert(3);
    uniqueNumbers.insert(7);
    
    // Iterate through set (automatically sorted)
    cout << "Set elements: ";
    for (const auto& num : uniqueNumbers) {
        cout << num << " ";
    }
    cout << endl;
    
    // Check if element exists
    if (uniqueNumbers.find(5) != uniqueNumbers.end()) {
        cout << "5 is in the set" << endl;
    }
    
    // ALGORITHMS
    cout << "\n=== ALGORITHMS ===" << endl;
    vector values = {5, 2, 8, 1, 9, 3, 7, 4, 6};
    
    // Find element
    auto it = find(values.begin(), values.end(), 7);
    if (it != values.end()) {
        cout << "Found 7 at position: " << distance(values.begin(), it) << endl;
    }
    
    // Count elements
    int countFives = count(values.begin(), values.end(), 5);
    cout << "Number of 5s: " << countFives << endl;
    
    // Min and max elements
    auto minIt = min_element(values.begin(), values.end());
    auto maxIt = max_element(values.begin(), values.end());
    cout << "Min element: " << *minIt << endl;
    cout << "Max element: " << *maxIt << endl;
    
    // Lambda function with algorithm
    cout << "Even numbers: ";
    for_each(values.begin(), values.end(), [](int n) {
        if (n % 2 == 0) cout << n << " ";
    });
    cout << endl;
    
    return 0;
}
