Variables and Data Types

Basic variable declarations in C++.

#include <iostream>
using namespace std;

int main() {
    // Integer types
    int age = 25;
    long bigNumber = 123456789L;
    
    // Floating point
    float price = 19.99f;
    double pi = 3.1415926535;
    
    // Character
    char grade = 'A';
    
    // Boolean
    bool isCppFun = true;
    
    // String
    string name = "Alice";
    
    cout << "Name: " << name << endl;
    return 0;
}

Control Structures

Conditionals and loops in C++.

#include <iostream>
using namespace std;

int main() {
    // If-else statement
    int x = 10;
    if (x > 5) {
        cout << "x is greater than 5" << endl;
    } else {
        cout << "x is 5 or less" << endl;
    }
    
    // For loop
    for (int i = 0; i < 5; i++) {
        cout << "Iteration " << i << endl;
    }
    
    // While loop
    int count = 3;
    while (count > 0) {
        cout << count << endl;
        count--;
    }
    
    return 0;
}

Functions

Creating and using functions in C++.

#include <iostream>
using namespace std;

// Function declaration
int addNumbers(int a, int b);

int main() {
    int num1 = 5, num2 = 7;
    int sum = addNumbers(num1, num2);
    
    cout << "Sum: " << sum << endl;
    return 0;
}

// Function definition
int addNumbers(int a, int b) {
    return a + b;
}

Classes and Objects

Basic OOP in C++.

#include <iostream>
using namespace std;

class Rectangle {
  private:
    int width, height;
  
  public:
    Rectangle(int w, int h) {
        width = w;
        height = h;
    }
    
    int area() {
        return width * height;
    }
};

int main() {
    Rectangle rect(3, 4);
    cout << "Area: " << rect.area() << endl;
    return 0;
}
