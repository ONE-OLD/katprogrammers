Basic HTML Structure

The foundation of every HTML document.

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document Title</title>
    <meta name="description" content="Page description">
</head>
<body>
    <h1>Main Heading</h1>
    <p>This is a paragraph of text.</p>
    <ul>
        <li>List item 1</li>
        <li>List item 2</li>
    </ul>
</body>
</html>

Common HTML Elements

Essential elements every developer should know.

<!-- Headings -->
<h1>Main Title</h1>
<h2>Subheading</h2>
<h3>Section Heading</h3>

<!-- Text elements -->
<p>This is a paragraph. <strong>Important text</strong> and <em>emphasized text</em>.</p>
<p>Here's a <a href="https://example.com">link</a> to another page.</p>

<!-- Lists -->
<ul>
    <li>Unordered list item</li>
    <li>Another item</li>
</ul>

<ol>
    <li>First ordered item</li>
    <li>Second item</li>
</ol>

<!-- Images -->
<img src="image.jpg" alt="Description of image" width="400" height="300">

<!-- Divisions -->
<div class="container">
    <p>Content grouped together</p>
</div>

HTML Forms

Collecting user input with forms.

<form action="/submit" method="post">
    <fieldset>
        <legend>Personal Information</legend>
        
        <div>
            <label for="name">Full Name:</label>
            <input type="text" id="name" name="name" required>
        </div>
        
        <div>
            <label for="email">Email:</label>
            <input type="email" id="email" name="email" required>
        </div>
        
        <div>
            <label for="password">Password:</label>
            <input type="password" id="password" name="password" minlength="8" required>
        </div>
    </fieldset>
    
    <fieldset>
        <legend>Preferences</legend>
        
        <div>
            <label for="newsletter">
                <input type="checkbox" id="newsletter" name="newsletter" checked>
                Subscribe to newsletter
            </label>
        </div>
        
        <div>
            <label>Age Group:</label>
            <select name="age">
                <option value="under18">Under 18</option>
                <option value="18-25" selected>18-25</option>
                <option value="26-35">26-35</option>
                <option value="36+">36+</option>
            </select>
        </div>
        
        <div>
            <label>Feedback:</label>
            <textarea name="feedback" rows="4"></textarea>
        </div>
    </fieldset>
    
    <button type="submit">Submit Form</button>
    <button type="reset">Reset</button>
</form>

Semantic HTML5

Modern semantic elements for better structure and accessibility.

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Semantic HTML Example</title>
</head>
<body>
    <header>
        <h1>Website Title</h1>
        <nav>
            <ul>
                <li><a href="#home">Home</a></li>
                <li><a href="#about">About</a></li>
                <li><a href="#contact">Contact</a></li>
            </ul>
        </nav>
    </header>
    
    <main>
        <article>
            <header>
                <h2>Article Title</h2>
                <p>Posted on <time datetime="2023-05-15">May 15, 2023</time></p>
            </header>
            
            <section>
                <h3>Introduction</h3>
                <p>Article introduction content...</p>
            </section>
            
            <section>
                <h3>Main Content</h3>
                <p>Detailed article content...</p>
                <figure>
                    <img src="image.jpg" alt="Description">
                    <figcaption>Image caption</figcaption>
                </figure>
            </section>
            
            <footer>
                <p>Article footer with author info</p>
            </footer>
        </article>
        
        <aside>
            <h2>Related Content</h2>
            <nav>
                <ul>
                    <li><a href="#">Related Article 1</a></li>
                    <li><a href="#">Related Article 2</a></li>
                </ul>
            </nav>
        </aside>
    </main>
    
    <footer>
        <p>© 2023 Company Name. All rights reserved.</p>
        <address>
            Contact us at <a href="mailto:info@example.com">info@example.com</a>
        </address>
    </footer>
</body>
</html>
