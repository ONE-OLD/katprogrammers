MySQL Basics

Fundamental MySQL commands and operations.

-- Create a new database
CREATE DATABASE company_db;
USE company_db;

-- Create tables
CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE,
    hire_date DATE,
    salary DECIMAL(10, 2),
    department_id INT
);

CREATE TABLE departments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    location VARCHAR(100)
);

CREATE TABLE projects (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    budget DECIMAL(15, 2),
    start_date DATE,
    end_date DATE
);

CREATE TABLE employee_projects (
    employee_id INT,
    project_id INT,
    role VARCHAR(50),
    PRIMARY KEY (employee_id, project_id),
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    FOREIGN KEY (project_id) REFERENCES projects(id)
);

-- Basic SELECT queries
SELECT * FROM employees;
SELECT first_name, last_name, salary FROM employees;
SELECT first_name, last_name FROM employees WHERE salary > 70000;
SELECT * FROM employees WHERE hire_date >= '2020-01-01';

-- UPDATE statements
UPDATE employees SET salary = 80000.00 WHERE id = 1;
UPDATE employees SET salary = salary * 1.1 WHERE department_id = 1;

-- DELETE statements
DELETE FROM employees WHERE id = 5;

-- ALTER TABLE examples
ALTER TABLE employees ADD phone_number VARCHAR(15);
ALTER TABLE employees MODIFY salary DECIMAL(12, 2);
ALTER TABLE employees DROP COLUMN phone_number;

-- DROP TABLE
DROP TABLE IF EXISTS temporary_data;

-- TRUNCATE TABLE (removes all data but keeps structure)
TRUNCATE TABLE employee_projects;

SELECT Queries

Advanced SELECT query techniques with filtering, sorting, and aggregation.

-- Basic SELECT with filtering
SELECT * FROM employees WHERE department_id = 1;
SELECT first_name, last_name FROM employees WHERE salary BETWEEN 60000 AND 80000;
SELECT * FROM employees WHERE last_name LIKE 'S%';
SELECT * FROM employees WHERE hire_date BETWEEN '2020-01-01' AND '2021-12-31';

-- Sorting results
SELECT * FROM employees ORDER BY last_name ASC;
SELECT * FROM employees ORDER BY salary DESC;
SELECT * FROM employees ORDER BY department_id ASC, salary DESC;

-- Limiting results
SELECT * FROM employees LIMIT 5;
SELECT * FROM employees ORDER BY hire_date DESC LIMIT 3;
SELECT * FROM employees LIMIT 5 OFFSET 2; -- Skip first 2, get next 5

-- Aggregate functions
SELECT COUNT(*) AS total_employees FROM employees;
SELECT AVG(salary) AS average_salary FROM employees;
SELECT MAX(salary) AS highest_salary, MIN(salary) AS lowest_salary FROM employees;
SELECT SUM(salary) AS total_salary_budget FROM employees;

-- GROUP BY with aggregates
SELECT department_id, COUNT(*) AS employee_count 
FROM employees 
GROUP BY department_id;

SELECT department_id, AVG(salary) AS avg_salary 
FROM employees 
GROUP BY department_id 
HAVING AVG(salary) > 70000;

SELECT YEAR(hire_date) AS hire_year, COUNT(*) AS hires 
FROM employees 
GROUP BY YEAR(hire_date) 
ORDER BY hire_year;

-- String functions
SELECT CONCAT(first_name, ' ', last_name) AS full_name FROM employees;
SELECT UPPER(first_name), LOWER(last_name) FROM employees;
SELECT LENGTH(first_name) AS name_length FROM employees;
SELECT SUBSTRING(email, 1, 5) AS email_prefix FROM employees;

-- Date functions
SELECT first_name, last_name, DATEDIFF(CURDATE(), hire_date) AS days_employed 
FROM employees;

SELECT first_name, last_name, 
       YEAR(hire_date) AS hire_year,
       MONTH(hire_date) AS hire_month,
       DAY(hire_date) AS hire_day
FROM employees;

SELECT * FROM employees 
WHERE YEAR(hire_date) = 2020;

-- Mathematical functions
SELECT first_name, last_name, salary, 
       ROUND(salary * 0.1, 2) AS bonus 
FROM employees;

SELECT first_name, last_name, 
       salary, 
       salary * 12 AS annual_salary 
FROM employees;

-- Conditional logic with CASE
SELECT first_name, last_name, salary,
       CASE 
           WHEN salary >= 80000 THEN 'High'
           WHEN salary >= 60000 THEN 'Medium'
           ELSE 'Low'
       END AS salary_grade
FROM employees;

SELECT first_name, last_name, department_id,
       CASE department_id
           WHEN 1 THEN 'Engineering'
           WHEN 2 THEN 'Marketing'
           WHEN 3 THEN 'Sales'
           WHEN 4 THEN 'HR'
           ELSE 'Other'
       END AS department_name
FROM employees;

-- DISTINCT values
SELECT DISTINCT department_id FROM employees;
SELECT COUNT(DISTINCT department_id) AS unique_departments FROM employees;

-- Combining conditions
SELECT * FROM employees 
WHERE (department_id = 1 AND salary > 70000) 
   OR (department_id = 2 AND salary > 65000);

SELECT * FROM employees 
WHERE department_id IN (1, 3) 
AND hire_date >= '2020-01-01';

-- NULL handling
SELECT * FROM employees WHERE email IS NULL;
SELECT * FROM employees WHERE email IS NOT NULL;

-- COALESCE for handling NULLs
SELECT first_name, last_name, 
       COALESCE(email, 'No email provided') AS email_address 
FROM employees;

JOIN Operations

Combining data from multiple tables using various JOIN operations.

-- INNER JOIN (most common)
SELECT e.first_name, e.last_name, d.name AS department_name
FROM employees e
INNER JOIN departments d ON e.department_id = d.id;

-- LEFT JOIN (all employees, even if no department)
SELECT e.first_name, e.last_name, d.name AS department_name
FROM employees e
LEFT JOIN departments d ON e.department_id = d.id;

-- RIGHT JOIN (all departments, even if no employees)
SELECT e.first_name, e.last_name, d.name AS department_name
FROM employees e
RIGHT JOIN departments d ON e.department_id = d.id;

-- FULL OUTER JOIN (MySQL doesn't support directly, emulate with UNION)
SELECT e.first_name, e.last_name, d.name AS department_name
FROM employees e
LEFT JOIN departments d ON e.department_id = d.id
UNION
SELECT e.first_name, e.last_name, d.name AS department_name
FROM employees e
RIGHT JOIN departments d ON e.department_id = d.id;

-- Multiple JOINs
SELECT e.first_name, e.last_name, d.name AS department_name, p.name AS project_name
FROM employees e
INNER JOIN departments d ON e.department_id = d.id
INNER JOIN employee_projects ep ON e.id = ep.employee_id
INNER JOIN projects p ON ep.project_id = p.id;

-- Self JOIN (employees and their managers example)
-- First, alter table to add manager_id
ALTER TABLE employees ADD manager_id INT;
UPDATE employees SET manager_id = 2 WHERE id = 1;
UPDATE employees SET manager_id = 4 WHERE id = 3;

SELECT e1.first_name AS employee_name, 
       e1.last_name AS employee_last_name,
       e2.first_name AS manager_name,
       e2.last_name AS manager_last_name
FROM employees e1
LEFT JOIN employees e2 ON e1.manager_id = e2.id;

-- CROSS JOIN (Cartesian product)
SELECT e.first_name, d.name
FROM employees e
CROSS JOIN departments d;

-- NATURAL JOIN (joins on columns with same name)
-- Note: Use with caution as it can produce unexpected results
SELECT e.first_name, d.name
FROM employees e
NATURAL JOIN departments d;

-- JOIN with aggregate functions
SELECT d.name AS department_name, 
       COUNT(e.id) AS employee_count,
       AVG(e.salary) AS average_salary
FROM departments d
LEFT JOIN employees e ON d.id = e.department_id
GROUP BY d.id, d.name
ORDER BY average_salary DESC;

-- JOIN with WHERE clause
SELECT e.first_name, e.last_name, d.name AS department_name
FROM employees e
INNER JOIN departments d ON e.department_id = d.id
WHERE d.location = 'New York';

-- JOIN with multiple conditions
SELECT e.first_name, e.last_name, p.name AS project_name, ep.role
FROM employees e
INNER JOIN employee_projects ep ON e.id = ep.employee_id
INNER JOIN projects p ON ep.project_id = p.id
WHERE p.budget > 50000 AND ep.role = 'Developer';

-- Using table aliases
SELECT emp.first_name, emp.last_name, dept.name AS department
FROM employees emp
JOIN departments dept ON emp.department_id = dept.id;

-- JOIN with subquery
SELECT e.first_name, e.last_name, dept_stats.avg_salary
FROM employees e
JOIN (
    SELECT department_id, AVG(salary) AS avg_salary
    FROM employees
    GROUP BY department_id
) dept_stats ON e.department_id = dept_stats.department_id
WHERE e.salary > dept_stats.avg_salary;

-- Multiple JOIN types in one query
SELECT e.first_name, e.last_name, 
       d.name AS department_name,
       p.name AS project_name,
       m.first_name AS manager_name
FROM employees e
LEFT JOIN departments d ON e.department_id = d.id
LEFT JOIN employee_projects ep ON e.id = ep.employee_id
LEFT JOIN projects p ON ep.project_id = p.id
LEFT JOIN employees m ON e.manager_id = m.id;

Transactions and ACID Properties

Managing database transactions with ACID compliance.

-- Start a transaction
START TRANSACTION;

-- Perform multiple operations
UPDATE accounts SET balance = balance - 1000 WHERE account_id = 123;
UPDATE accounts SET balance = balance + 1000 WHERE account_id = 456;

-- Commit the transaction if both operations succeed
COMMIT;

-- Example with error handling
START TRANSACTION;

BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SELECT 'Transaction failed: ' + ERROR_MESSAGE();
    END;

    -- Transfer money between accounts
    UPDATE accounts SET balance = balance - 1000 WHERE account_id = 123;
    UPDATE accounts SET balance = balance + 1000 WHERE account_id = 456;
    
    COMMIT;
    SELECT 'Transaction completed successfully';
END;

-- Savepoints within transactions
START TRANSACTION;

UPDATE accounts SET balance = balance - 500 WHERE account_id = 123;
SAVEPOINT savepoint1;

UPDATE accounts SET balance = balance + 500 WHERE account_id = 456;
-- If something goes wrong, we can rollback to savepoint1
-- ROLLBACK TO savepoint1;

COMMIT;

-- Transaction isolation levels
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
START TRANSACTION;
SELECT * FROM accounts WHERE account_id = 123;
COMMIT;

-- SET TRANSACTION ISOLATION LEVEL REPEATABLE READ;
-- SET TRANSACTION ISOLATION LEVEL SERIALIZABLE;

-- Locking within transactions
START TRANSACTION;
SELECT * FROM accounts WHERE account_id = 123 FOR UPDATE;
-- Other transactions cannot modify this row until we commit
UPDATE accounts SET balance = balance - 200 WHERE account_id = 123;
COMMIT;

-- Deadlock handling example
-- Transaction 1
START TRANSACTION;
UPDATE accounts SET balance = balance - 100 WHERE account_id = 123;
-- Transaction 2 (in another session)
START TRANSACTION;
UPDATE accounts SET balance = balance - 100 WHERE account_id = 456;
-- Transaction 1 then tries:
UPDATE accounts SET balance = balance + 100 WHERE account_id = 456;
-- Transaction 2 then tries:
UPDATE accounts SET balance = balance + 100 WHERE account_id = 123;
-- This would cause a deadlock

-- Using LOCK TABLES (less common, use with caution)
LOCK TABLES accounts WRITE;
UPDATE accounts SET balance = balance + 500 WHERE account_id = 123;
UNLOCK TABLES;

-- Transaction with multiple statements
START TRANSACTION;

INSERT INTO transactions (from_account, to_account, amount, transaction_date)
VALUES (123, 456, 1000, NOW());

UPDATE accounts SET balance = balance - 1000 WHERE account_id = 123;
UPDATE accounts SET balance = balance + 1000 WHERE account_id = 456;

COMMIT;

-- Rollback example
START TRANSACTION;

DELETE FROM employee_projects WHERE employee_id = 1;
-- Oops, that was a mistake!
ROLLBACK;

-- Nested transactions (MySQL doesn't support true nesting, but we can use savepoints)
START TRANSACTION;

UPDATE accounts SET balance = balance - 1000 WHERE account_id = 123;
SAVEPOINT nested_transaction;

UPDATE accounts SET balance = balance + 1000 WHERE account_id = 456;
-- If we need to rollback just the nested part:
-- ROLLBACK TO SAVEPOINT nested_transaction;

COMMIT;

-- Transaction with error checking in stored procedure
DELIMITER //
CREATE PROCEDURE transfer_money(
    IN from_account INT, 
    IN to_account INT, 
    IN amount DECIMAL(10,2)
)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SELECT 'Transfer failed' AS result;
    END;
    
    START TRANSACTION;
    
    -- Check if sufficient balance
    IF (SELECT balance FROM accounts WHERE account_id = from_account) < amount THEN
        ROLLBACK;
        SELECT 'Insufficient balance' AS result;
    ELSE
        UPDATE accounts SET balance = balance - amount WHERE account_id = from_account;
        UPDATE accounts SET balance = balance + amount WHERE account_id = to_account;
        
        INSERT INTO transaction_log (from_account, to_account, amount, status)
        VALUES (from_account, to_account, amount, 'COMPLETED');
        
        COMMIT;
        SELECT 'Transfer successful' AS result;
    END IF;
END //
DELIMITER ;
